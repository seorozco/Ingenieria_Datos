# 1. IMPORTAR Y CONECTAR (3 líneas)
import sys
sys.path.insert(0, '..')
from conexion_sqlserver import obtener_engines, SERVIDOR, BD_ORIGEN, BD_DESTINO

engine_origen, engine_destino = obtener_engines()

# 2. LEER DATOS DEL ORIGEN
import pandas as pd

df = pd.read_sql_query("""
    SELECT TOP 10 * FROM Sales.Customers
""", engine_origen)

# 3. ESCRIBIR AL DESTINO
df.to_sql('tabla_nombre', engine_destino, schema='dw', if_exists='append', index=False)

# 4. EJECUTAR SQL EN DESTINO
from sqlalchemy import text

with engine_destino.begin() as conn:
    conn.execute(text("""
        CREATE SCHEMA IF NOT EXISTS dw
    """))

# 5. LEER DESDE DESTINO (VERIFICACIÓN)
df_check = pd.read_sql_query("""
    SELECT * FROM dw.tabla_nombre
""", engine_destino)

# 6. INFORMACIÓN DISPONIBLE
print(f"Servidor: {SERVIDOR}")
print(f"Origen: {BD_ORIGEN}")
print(f"Destino: {BD_DESTINO}")

# === CASOS COMUNES ===

# A. Lectura con filtros
df = pd.read_sql_query("""
    SELECT * FROM Sales.Customers 
    WHERE CityID = 1
    ORDER BY CustomerID
""", engine_origen)

# B. Inserción con control de duplicados
df.to_sql('clientes', engine_destino, schema='dw', 
          if_exists='append', index=False)  # append = sin borrar existentes

# C. Transacción múltiple
with engine_destino.begin() as conn:
    conn.execute(text("DELETE FROM dw.tabla_temp"))
    # ... más operaciones
    # Se confirma automáticamente al salir del contexto

# D. Consulta sin pandas (resultado directo)
with engine_destino.connect() as conn:
    rows = conn.execute(text("SELECT COUNT(*) FROM dw.tabla")).fetchall()
    print(f"Total: {rows[0][0]}")
